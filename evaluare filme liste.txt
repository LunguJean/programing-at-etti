#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

typedef struct Utilizator {
    char nume[21];
    float scoruri[5];
    struct Utilizator *next;
} Utilizator;

// Creează un utilizator
Utilizator *create_user() {
    Utilizator *nou = malloc(sizeof(Utilizator));
    scanf("%s", nou->nume);
    for (int i = 0; i < 5; i++)
        scanf("%f", &nou->scoruri[i]);
    nou->next = NULL;
    return nou;
}

// Adaugă utilizator în listă
Utilizator *add_user(Utilizator *head) {
    if (head == NULL)
        return create_user();
    Utilizator *temp = head;
    while (temp->next)
        temp = temp->next;
    temp->next = create_user();
    return head;
}

// Afișare utilizatori (comanda 2)
void afisare(Utilizator *head) {
    while (head) {
        printf("%s ", head->nume);
        for (int i = 0; i < 5; i++)
            printf("%.0f ", head->scoruri[i]);
        printf("\n");
        head = head->next;
    }
}

// Media scorurilor
float medie(Utilizator *u) {
    float s = 0;
    for (int i = 0; i < 5; i++)
        s += u->scoruri[i];
    return s / 5.0;
}

// Comanda 3 — utilizatorul cel mai critic
void cel_mai_critic(Utilizator *head) {
    if (!head) return;
    Utilizator *min = head;
    float min_medie = medie(head);
    head = head->next;
    while (head) {
        float m = medie(head);
        if (m < min_medie) {
            min_medie = m;
            min = head;
        }
        head = head->next;
    }
    printf("%s ", min->nume);
    for (int i = 0; i < 5; i++)
        printf("%.0f ", min->scoruri[i]);
    printf("\n");
}

// Comanda 4 — cei mai similari utilizatori
void gusturi_similare(Utilizator *head) {
    Utilizator *v[100];
    int n = 0;
    while (head) {
        v[n++] = head;
        head = head->next;
    }

    double min_dist = 1e9;
    int p1 = 0, p2 = 1;

    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            double d = 0;
            for (int k = 0; k < 5; k++)
                d += pow(v[i]->scoruri[k] - v[j]->scoruri[k], 2);
            d = sqrt(d);
            if (d < min_dist) {
                min_dist = d;
                p1 = i; p2 = j;
            }
        }
    }

    printf("%s ", v[p1]->nume);
    for (int i = 0; i < 5; i++)
        printf("%.0f ", v[p1]->scoruri[i]);
    printf("\n");
    printf("%s ", v[p2]->nume);
    for (int i = 0; i < 5; i++)
        printf("%.0f ", v[p2]->scoruri[i]);
    printf("\n");
}

// Comanda 5 — top 3 filme
void top3_filme(Utilizator *head) {
    Utilizator *v[100];
    int n = 0;
    while (head) {
        v[n++] = head;
        head = head->next;
    }

    float medii[5];
    for (int f = 0; f < 5; f++) {
        float min = 1e9, max = -1e9, sum = 0;
        for (int i = 0; i < n; i++) {
            float val = v[i]->scoruri[f];
            if (val < min) min = val;
            if (val > max) max = val;
            sum += val;
        }
        medii[f] = (sum - min - max) / (n - 2);
    }

    int idx[5] = {0, 1, 2, 3, 4};
    for (int i = 0; i < 4; i++)
        for (int j = i + 1; j < 5; j++)
            if (medii[idx[i]] < medii[idx[j]]) {
                int aux = idx[i];
                idx[i] = idx[j];
                idx[j] = aux;
            }

    printf("%d %d %d\n", idx[0], idx[1], idx[2]);
}

int main() {
    Utilizator *head = NULL;
    int comanda;

    while (scanf("%d", &comanda) != EOF) {
        if (comanda == 1)
            head = add_user(head);
        else if (comanda == 2) {
            afisare(head);
            break;
        } else if (comanda == 3) {
            cel_mai_critic(head);
            break;
        } else if (comanda == 4) {
            gusturi_similare(head);
            break;
        } else if (comanda == 5) {
            top3_filme(head);
            break;
        }
    }

    while (head) {
        Utilizator *aux = head;
        head = head->next;
        free(aux);
    }

    return 0;
}
