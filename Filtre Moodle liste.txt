#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Student {
    char nume[21];
    int grupa;
    int n;
    int note[20];
    float medie;
    struct Student *next;
} Student;

// Calculează media unui student
float calculeaza_medie(Student *s) {
    float suma = 0;
    for (int i = 0; i < s->n; i++)
        suma += s->note[i];
    return suma / s->n;
}

// Creează un student nou
Student *create_student() {
    Student *nou = malloc(sizeof(Student));
    scanf("%s", nou->nume);
    scanf("%d", &nou->grupa);
    scanf("%d", &nou->n);
    for (int i = 0; i < nou->n; i++)
        scanf("%d", &nou->note[i]);
    nou->medie = calculeaza_medie(nou);
    nou->next = NULL;
    return nou;
}

// Adaugă student în listă
Student *add_student(Student *head) {
    if (head == NULL)
        return create_student();
    Student *temp = head;
    while (temp->next)
        temp = temp->next;
    temp->next = create_student();
    return head;
}

// Afișare listă (comanda 2)
void afisare(Student *head) {
    while (head) {
        printf("%s %d ", head->nume, head->grupa);
        for (int i = 0; i < head->n; i++)
            printf("%d ", head->note[i]);
        printf("\n");
        head = head->next;
    }
}

// Afișează studentul cu media maximă (comanda 3)
void medie_maxima(Student *head) {
    Student *max = head;
    Student *temp = head->next;
    while (temp) {
        if (temp->medie > max->medie)
            max = temp;
        temp = temp->next;
    }
    printf("%s %d ", max->nume, max->grupa);
    for (int i = 0; i < max->n; i++)
        printf("%d ", max->note[i]);
    printf("\n");
}

// Filtrare după grupă (comanda 4)
Student *filtrare_grupa(Student *head, int grupa) {
    Student *nou = NULL, *tail = NULL;
    while (head) {
        if (head->grupa == grupa) {
            Student *copie = malloc(sizeof(Student));
            *copie = *head;
            copie->next = NULL;
            if (nou == NULL)
                nou = tail = copie;
            else {
                tail->next = copie;
                tail = copie;
            }
        }
        head = head->next;
    }
    return nou;
}

// Sortare după medie crescător (comanda 5)
Student *sortare_grupa(Student *head, int grupa) {
    // creăm lista filtrată
    Student *lista = filtrare_grupa(head, grupa);
    if (!lista) return NULL;

    // transformăm în vector pentru sortare simplă
    Student *v[100];
    int n = 0;
    Student *temp = lista;
    while (temp) {
        v[n++] = temp;
        temp = temp->next;
    }

    for (int i = 0; i < n - 1; i++)
        for (int j = i + 1; j < n; j++)
            if (v[i]->medie > v[j]->medie) {
                Student *aux = v[i];
                v[i] = v[j];
                v[j] = aux;
            }

    // reconstruim lista sortată
    for (int i = 0; i < n - 1; i++)
        v[i]->next = v[i + 1];
    v[n - 1]->next = NULL;

    return v[0];
}

int main() {
    Student *head = NULL;
    int comanda;

    while (scanf("%d", &comanda) != EOF) {
        if (comanda == 1) {
            head = add_student(head);
        } else if (comanda == 2) {
            afisare(head);
        } else if (comanda == 3) {
            medie_maxima(head);
        } else if (comanda == 4) {
            int grupa;
            scanf("%d", &grupa);
            Student *lista = filtrare_grupa(head, grupa);
            afisare(lista);
            while (lista) { Student *aux = lista; lista = lista->next; free(aux); }
        } else if (comanda == 5) {
            int grupa;
            scanf("%d", &grupa);
            Student *lista = sortare_grupa(head, grupa);
            afisare(lista);
            while (lista) { Student *aux = lista; lista = lista->next; free(aux); }
        }
    }

    while (head) {
        Student *aux = head;
        head = head->next;
        free(aux);
    }

    return 0;
}
