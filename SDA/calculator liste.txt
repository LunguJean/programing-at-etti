#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Nod {
    int cifra;
    struct Nod *next;
} Nod;

// Creează o listă dintr-un șir numeric
Nod *creare_lista(char numar[]) {
    Nod *head = NULL, *tail = NULL;
    for (int i = 0; numar[i]; i++) {
        Nod *nou = malloc(sizeof(Nod));
        nou->cifra = numar[i] - '0';
        nou->next = NULL;
        if (!head)
            head = tail = nou;
        else {
            tail->next = nou;
            tail = nou;
        }
    }
    return head;
}

// Afișează lista
void afisare(Nod *head) {
    while (head) {
        printf("%d", head->cifra);
        head = head->next;
    }
    printf("\n");
}

// Inversează lista (pentru adunare/scădere)
Nod *inversare(Nod *head) {
    Nod *prev = NULL, *cur = head, *next;
    while (cur) {
        next = cur->next;
        cur->next = prev;
        prev = cur;
        cur = next;
    }
    return prev;
}

// Adună două numere mari
Nod *adunare(Nod *a, Nod *b) {
    a = inversare(a);
    b = inversare(b);

    Nod *rez = NULL, *tail = NULL;
    int carry = 0;

    while (a || b || carry) {
        int suma = carry;
        if (a) { suma += a->cifra; a = a->next; }
        if (b) { suma += b->cifra; b = b->next; }
        carry = suma / 10;
        suma %= 10;

        Nod *nou = malloc(sizeof(Nod));
        nou->cifra = suma;
        nou->next = NULL;

        if (!rez) rez = tail = nou;
        else { tail->next = nou; tail = nou; }
    }

    return inversare(rez);
}

// Scade două numere mari (a - b), presupunem a ≥ b
Nod *scadere(Nod *a, Nod *b, int *semn) {
    // convertim în vectori pentru a compara mărimea
    char sa[1000] = "", sb[1000] = "";
    Nod *temp = a; int ia = 0;
    while (temp) { sa[ia++] = temp->cifra + '0'; temp = temp->next; }
    temp = b; int ib = 0;
    while (temp) { sb[ib++] = temp->cifra + '0'; temp = temp->next; }
    sa[ia] = sb[ib] = '\0';

    if (strlen(sa) < strlen(sb) || (strlen(sa) == strlen(sb) && strcmp(sa, sb) < 0)) {
        *semn = -1;
        Nod *aux = a; a = b; b = aux;
    } else *semn = 1;

    a = inversare(a);
    b = inversare(b);

    Nod *rez = NULL, *tail = NULL;
    int imprumut = 0;

    while (a) {
        int x = a->cifra - imprumut;
        int y = (b ? b->cifra : 0);
        if (x < y) { x += 10; imprumut = 1; } else imprumut = 0;

        int dif = x - y;

        Nod *nou = malloc(sizeof(Nod));
        nou->cifra = dif;
        nou->next = NULL;

        if (!rez) rez = tail = nou;
        else { tail->next = nou; tail = nou; }

        a = a->next;
        if (b) b = b->next;
    }

    // eliminăm zerourile de la început
    rez = inversare(rez);
    while (rez && rez->cifra == 0 && rez->next) {
        Nod *aux = rez;
        rez = rez->next;
        free(aux);
    }
    return rez;
}

// Verifică dacă un număr (listă) este palindrom
int palindrom(Nod *head) {
    int v[1000], n = 0;
    while (head) {
        v[n++] = head->cifra;
        head = head->next;
    }
    for (int i = 0; i < n / 2; i++)
        if (v[i] != v[n - 1 - i])
            return 0;
    return 1;
}

int main() {
    char num1[1001], num2[1001];
    scanf("%s", num1);
    scanf("%s", num2);

    Nod *a = creare_lista(num1);
    Nod *b = creare_lista(num2);

    int c;
    scanf("%d", &c);

    switch (c) {
        case 1:
            afisare(a);
            afisare(b);
            break;
        case 2: {
            Nod *suma = adunare(a, b);
            afisare(suma);
            break;
        }
        case 3: {
            int semn;
            Nod *dif = scadere(a, b, &semn);
            if (semn == -1) printf("-");
            afisare(dif);
            break;
        }
        case 4: {
            Nod *suma = adunare(a, b);
            if (palindrom(suma))
                printf("da\n");
            else
                printf("nu\n");
            break;
        }
    }

    return 0;
}
