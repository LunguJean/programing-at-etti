#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Pasager {
    char nume[30];
    char prenume[30];
    int rezervare;  // 0 dacă nu are rezervare, altfel numărul locului
    int loc;        // locul alocat
    struct Pasager *next;
} Pasager;

// ---- Funcții auxiliare ----

// Creare pasager
Pasager *create_pasager(int with_rezervare) {
    Pasager *nou = malloc(sizeof(Pasager));
    scanf("%s %s", nou->nume, nou->prenume);
    nou->loc = 0;
    nou->next = NULL;
    nou->rezervare = 0;
    if (with_rezervare)
        scanf("%d", &nou->rezervare);
    return nou;
}

// Adăugare pasager în listă
Pasager *add_pasager(Pasager *head, int with_rezervare) {
    if (head == NULL)
        return create_pasager(with_rezervare);
    Pasager *temp = head;
    while (temp->next)
        temp = temp->next;
    temp->next = create_pasager(with_rezervare);
    return head;
}

// Afișare pasageri (Comanda 1)
void afisare_pers(Pasager *head) {
    while (head) {
        printf("%s %s\n", head->nume, head->prenume);
        head = head->next;
    }
}

// ---- Comanda 2: Cel mai bun avion ----
void best_plane(Pasager *head) {
    int persoane = 0;
    while (head) {
        persoane++;
        head = head->next;
    }

    int best_x = 0, best_y = 0;
    int min_dif = 1e9;

    for (int i = 0; i < 4; i++) {
        int x, y;
        scanf("%d %d", &x, &y);
        int cap = x * y;
        if (cap >= persoane && cap - persoane < min_dif) {
            min_dif = cap - persoane;
            best_x = x;
            best_y = y;
        }
    }
    printf("%d %d\n", best_x, best_y);
}

// ---- Comanda 3: Echilibrare avion ----
void alocare_echilibru(Pasager *head, int x, int y) {
    int total = x * y;
    int *locuri = malloc(total * sizeof(int));
    int index = 0;

    // 1️⃣ locuri la fereastră (alternativ față/spate)
    int front = 1, back = total;
    while (front <= back) {
        locuri[index++] = front;
        if (index < total)
            locuri[index++] = back;
        front += x;
        back -= x;
    }

    // 2️⃣ restul locurilor, crescător
    for (int i = 1; i <= total; i++) {
        int found = 0;
        for (int j = 0; j < index; j++)
            if (locuri[j] == i) found = 1;
        if (!found)
            locuri[index++] = i;
    }

    // 3️⃣ alocare pasageri
    int i = 0;
    Pasager *temp = head;
    while (temp && i < total) {
        temp->loc = locuri[i++];
        temp = temp->next;
    }

    // sortare după loc pentru afișare
    for (Pasager *p = head; p; p = p->next)
        for (Pasager *q = p->next; q; q = q->next)
            if (p->loc > q->loc) {
                int aux = p->loc;
                p->loc = q->loc;
                q->loc = aux;
                char t1[30], t2[30];
                strcpy(t1, p->nume);
                strcpy(t2, p->prenume);
                strcpy(p->nume, q->nume);
                strcpy(p->prenume, q->prenume);
                strcpy(q->nume, t1);
                strcpy(q->prenume, t2);
            }

    // afișare
    temp = head;
    while (temp) {
        printf("%s %s %d\n", temp->nume, temp->prenume, temp->loc);
        temp = temp->next;
    }

    free(locuri);
}

// ---- Comanda 4: Alocare cu rezervări ----
void alocare_rezervari(Pasager *head, int x, int y) {
    int total = x * y;
    int *ocupat = calloc(total + 1, sizeof(int));

    // alocăm locurile rezervate
    Pasager *temp = head;
    while (temp) {
        if (temp->rezervare != 0 && temp->rezervare <= total) {
            temp->loc = temp->rezervare;
            ocupat[temp->rezervare] = 1;
        }
        temp = temp->next;
    }

    // alocăm restul locurilor crescător
    temp = head;
    for (int i = 1; i <= total; i++) {
        if (ocupat[i] == 0) {
            while (temp && temp->loc != 0) temp = temp->next;
            if (!temp) break;
            temp->loc = i;
            ocupat[i] = 1;
        }
    }

    // sortare după loc
    for (Pasager *p = head; p; p = p->next)
        for (Pasager *q = p->next; q; q = q->next)
            if (p->loc > q->loc) {
                int aux = p->loc;
                p->loc = q->loc;
                q->loc = aux;
                char t1[30], t2[30];
                strcpy(t1, p->nume);
                strcpy(t2, p->prenume);
                strcpy(p->nume, q->nume);
                strcpy(p->prenume, q->prenume);
                strcpy(q->nume, t1);
                strcpy(q->prenume, t2);
            }

    // afișare
    temp = head;
    while (temp) {
        printf("%s %s %d\n", temp->nume, temp->prenume, temp->loc);
        temp = temp->next;
    }

    free(ocupat);
}

int main() {
    int x, y;
    scanf("%d %d", &x, &y);

    int c;
    scanf("%d", &c);

    Pasager *head = NULL;

    if (c == 1) {
        char nume[30];
        while (1) {
            scanf("%s", nume);
            if (strcmp(nume, "STOP") == 0) break;
            Pasager *nou = malloc(sizeof(Pasager));
            strcpy(nou->nume, nume);
            scanf("%s", nou->prenume);
            nou->next = NULL;
            nou->loc = 0;
            nou->rezervare = 0;
            if (!head) head = nou;
            else {
                Pasager *temp = head;
                while (temp->next) temp = temp->next;
                temp->next = nou;
            }
        }
        afisare_pers(head);
    }

    else if (c == 2) {
        char nume[30];
        while (1) {
            scanf("%s", nume);
            if (strcmp(nume, "STOP") == 0) break;
            head = add_pasager(head, 0);
        }
        best_plane(head);
    }

    else if (c == 3) {
        char nume[30];
        while (1) {
            scanf("%s", nume);
            if (strcmp(nume, "STOP") == 0) break;
            head = add_pasager(head, 0);
        }
        alocare_echilibru(head, x, y);
    }

    else if (c == 4) {
        char nume[30];
        while (1) {
            scanf("%s", nume);
            if (strcmp(nume, "STOP") == 0) break;
            head = add_pasager(head, 1);
        }
        alocare_rezervari(head, x, y);
    }

    // eliberare memorie
    while (head) {
        Pasager *aux = head;
        head = head->next;
        free(aux);
    }

    return 0;
}
