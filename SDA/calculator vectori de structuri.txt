#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct Numar {
    int cifre[1000];
    int len;
};

// Creează structura dintr-un șir
struct Numar creare(char numar[]) {
    struct Numar x;
    x.len = strlen(numar);
    for (int i = 0; i < x.len; i++)
        x.cifre[i] = numar[i] - '0';
    return x;
}

// Afișare
void afisare(struct Numar x) {
    for (int i = 0; i < x.len; i++)
        printf("%d", x.cifre[i]);
    printf("\n");
}

// Adunare
struct Numar adunare(struct Numar a, struct Numar b) {
    struct Numar rez;
    int i = a.len - 1, j = b.len - 1, k = 0, carry = 0, temp[2000];

    while (i >= 0 || j >= 0 || carry) {
        int s = carry;
        if (i >= 0) s += a.cifre[i--];
        if (j >= 0) s += b.cifre[j--];
        carry = s / 10;
        temp[k++] = s % 10;
    }

    rez.len = k;
    for (int i = 0; i < k; i++)
        rez.cifre[i] = temp[k - 1 - i];

    return rez;
}

// Scădere (a - b)
struct Numar scadere(struct Numar a, struct Numar b, int *semn) {
    *semn = 1;

    if (a.len < b.len || (a.len == b.len && strcmp((char*)a.cifre, (char*)b.cifre) < 0)) {
        *semn = -1;
        struct Numar aux = a; a = b; b = aux;
    }

    struct Numar rez;
    int i = a.len - 1, j = b.len - 1, k = 0, imprumut = 0, temp[2000];

    while (i >= 0) {
        int x = a.cifre[i] - imprumut;
        int y = (j >= 0) ? b.cifre[j] : 0;
        if (x < y) { x += 10; imprumut = 1; } else imprumut = 0;
        temp[k++] = x - y;
        i--; j--;
    }

    while (k > 1 && temp[k - 1] == 0) k--;
    rez.len = k;
    for (int i = 0; i < k; i++)
        rez.cifre[i] = temp[k - 1 - i];
    return rez;
}

// Palindrom
int palindrom(struct Numar x) {
    for (int i = 0; i < x.len / 2; i++)
        if (x.cifre[i] != x.cifre[x.len - 1 - i])
            return 0;
    return 1;
}

int main() {
    char s1[1000], s2[1000];
    scanf("%s", s1);
    scanf("%s", s2);

    struct Numar a = creare(s1);
    struct Numar b = creare(s2);

    int c;
    scanf("%d", &c);

    switch (c) {
        case 1:
            afisare(a);
            afisare(b);
            break;
        case 2: {
            struct Numar s = adunare(a, b);
            afisare(s);
            break;
        }
        case 3: {
            int semn;
            struct Numar d = scadere(a, b, &semn);
            if (semn == -1) printf("-");
            afisare(d);
            break;
        }
        case 4: {
            struct Numar s = adunare(a, b);
            printf(palindrom(s) ? "da\n" : "nu\n");
            break;
        }
    }

    return 0;
}
