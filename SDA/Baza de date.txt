#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ================= STRUCTURI ================= */

typedef struct User {
    char name[31];
    int y, m, d;
    int hash;              // suma ASCII a username-ului
    struct User *next;
} User;

typedef struct {
    User **buckets;
    int size;
} HashMap;

/* ================= UTILE ================= */

int ascii_sum(const char *s) {
    int sum = 0;
    for (int i = 0; s[i]; i++)
        sum += (unsigned char)s[i];
    return sum;
}

int is_major(int y, int m, int d) {
    // data de referinta: 09.01.2024
    int Y = 2024, M = 1, D = 9;

    int age = Y - y;
    if (m > M || (m == M && d > D))
        age--;

    return age > 18;
}

int similar(const char *a, const char *b) {
    if (strlen(a) != strlen(b)) return 0;

    int cnt[256] = {0};
    for (int i = 0; a[i]; i++) cnt[(unsigned char)a[i]]++;
    for (int i = 0; b[i]; i++) cnt[(unsigned char)b[i]]--;

    for (int i = 0; i < 256; i++)
        if (cnt[i] != 0) return 0;

    return 1;
}

/* ================= HASHMAP ================= */

HashMap *create_map(int n) {
    HashMap *m = malloc(sizeof(HashMap));
    m->size = n;
    m->buckets = calloc(n, sizeof(User*));
    return m;
}

void insert(HashMap *m, const char *name, int y, int mo, int d) {
    int h = ascii_sum(name);
    int idx = h % m->size;

    User *u = malloc(sizeof(User));
    strcpy(u->name, name);
    u->y = y; u->m = mo; u->d = d;
    u->hash = h;
    u->next = NULL;

    if (!m->buckets[idx]) {
        m->buckets[idx] = u;
    } else {
        User *p = m->buckets[idx];
        while (p->next) p = p->next;
        p->next = u;
    }
}

/* ================= CERINTA 1 ================= */

void cmd1(HashMap *m) {
    for (int i = 0; i < m->size; i++) {
        User *u = m->buckets[i];
        while (u) {
            if (is_major(u->y, u->m, u->d))
                printf("%s %d %d %d\n", u->name, u->y, u->m, u->d);
            else
                printf("%s Minor\n", u->name);
            u = u->next;
        }
    }
}

/* ================= CERINTA 2 ================= */

void cmd2(char **names, int n) {
    for (int q = 0; q < 3; q++) {
        char s[31];
        scanf("%30s", s);

        int found = 0;
        for (int i = 0; i < n; i++) {
            if (similar(names[i], s)) {
                if (found) printf(" ");
                printf("%s", names[i]);
                found = 1;
            }
        }

        if (!found)
            printf("Nu exista useri similari");

        printf("\n");
    }
}

/* ================= CERINTA 3 ================= */

int ok_dim(int *hashes, int n, int dim) {
    int *cnt = calloc(dim, sizeof(int));

    for (int i = 0; i < n; i++) {
        int idx = hashes[i] % dim;
        cnt[idx]++;
        if (cnt[idx] > 2) {
            free(cnt);
            return 0;
        }
    }

    free(cnt);
    return 1;
}

int find_min_dim(int *hashes, int n) {
    for (int dim = 1; ; dim++)
        if (ok_dim(hashes, n, dim))
            return dim;
}

/* ================= MAIN ================= */

int main() {
    int n;
    scanf("%d", &n);

    int users;
    scanf("%d", &users);

    HashMap *m = create_map(n);

    char **names = malloc(users * sizeof(char*));
    int *hashes = malloc(users * sizeof(int));

    for (int i = 0; i < users; i++) {
        char name[31];
        int y, mo, d;

        scanf("%30s %d %d %d", name, &y, &mo, &d);

        names[i] = malloc(31);
        strcpy(names[i], name);
        hashes[i] = ascii_sum(name);

        insert(m, name, y, mo, d);
    }

    int cmd;
    scanf("%d", &cmd);

    if (cmd == 1)
        cmd1(m);
    else if (cmd == 2)
        cmd2(names, users);
    else if (cmd == 3)
        printf("%d\n", find_min_dim(hashes, users));

    return 0;
}
