#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ================= STRUCTURI ================= */

typedef struct Node {
    char word[21];
    struct Node *next;
} Node;

typedef struct {
    Node **buckets;
    int size;
} HashMap;

/* ================= UTILE ================= */

int ascii_sum(const char *s) {
    int sum = 0;
    for (int i = 0; s[i]; i++)
        sum += (unsigned char)s[i];
    return sum;
}

int is_anagram(const char *a, const char *b) {
    if (strlen(a) != strlen(b)) return 0;

    int cnt[256] = {0};
    for (int i = 0; a[i]; i++) cnt[(unsigned char)a[i]]++;
    for (int i = 0; b[i]; i++) cnt[(unsigned char)b[i]]--;

    for (int i = 0; i < 256; i++)
        if (cnt[i] != 0) return 0;

    return 1;
}

/* ================= HASHMAP ================= */

HashMap *create_map(int m) {
    HashMap *h = malloc(sizeof(HashMap));
    h->size = m;
    h->buckets = calloc(m, sizeof(Node*));
    return h;
}

void insert(HashMap *h, const char *word) {
    int idx = ascii_sum(word) % h->size;

    Node *n = malloc(sizeof(Node));
    strcpy(n->word, word);
    n->next = NULL;

    if (!h->buckets[idx]) {
        h->buckets[idx] = n;
    } else {
        Node *p = h->buckets[idx];
        while (p->next) p = p->next;
        p->next = n;          // adăugare la SFÂRȘIT (chaining)
    }
}

/* ================= COMENZI ================= */

void cmd1(HashMap *h) {
    for (int i = 0; i < h->size; i++) {
        Node *p = h->buckets[i];
        while (p) {
            printf("%s %d\n", p->word, i);
            p = p->next;
        }
    }
}

void cmd2(HashMap *h) {
    char q[21];
    scanf("%20s", q);

    int idx = ascii_sum(q) % h->size;
    Node *p = h->buckets[idx];

    while (p) {
        if (is_anagram(p->word, q))
            printf("%s\n", p->word);
        p = p->next;
    }
}

void cmd3(char **words, int n) {
    int *has_anagram = calloc(n, sizeof(int));

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (i != j && is_anagram(words[i], words[j])) {
                has_anagram[i] = 1;
                break;
            }
        }
    }

    for (int i = 0; i < n; i++)
        if (!has_anagram[i])
            printf("%s\n", words[i]);

    free(has_anagram);
}

/* ================= MAIN ================= */

int main() {
    int m, n;
    scanf("%d", &m);
    scanf("%d", &n);

    HashMap *h = create_map(m);

    char **words = malloc(n * sizeof(char*));
    for (int i = 0; i < n; i++) {
        words[i] = malloc(21);
        scanf("%20s", words[i]);
        insert(h, words[i]);
    }

    int cmd;
    scanf("%d", &cmd);

    if (cmd == 1)
        cmd1(h);
    else if (cmd == 2)
        cmd2(h);
    else if (cmd == 3)
        cmd3(words, n);

    return 0;
}
